\section{Đóng góp của luận văn}
\begin{frame}{Đóng góp của luận văn}

\vspace{0.5em}
\begin{enumerate}\setlength{\itemsep}{12pt}
    \item[\textcolor{uetblue}{\textbf{1}}] \textbf{Kiến trúc PhaBERT-CNN}\\[3pt]
    DNABERT-2 + MKC (CNN kernel 3,5,7 + gộp trung bình có mặt nạ)\\
    {\small $\rightarrow$ Trực tiếp trên chuỗi DNA thô, không cần dự đoán gen}

    \item[\textcolor{uetblue}{\textbf{2}}] \textbf{Đánh giá toàn diện}\\[3pt]
    5-fold CV $\times$ 4 nhóm độ dài\\
    {\small $\rightarrow$ \textbf{82,01--91,38\%} accuracy, cải thiện \textbf{0,36--6,65\%} vs SOTA}

    \item[\textcolor{uetblue}{\textbf{3}}] \textbf{Ablation study (7 cấu hình)}\\[3pt]
    {\small $\rightarrow$ MKC đóng góp \textbf{+0,49 -- +4,12\%} accuracy}
\end{enumerate}

\end{frame}

\begin{frame}{Ablation Study -- Đóng góp của MKC}

\begin{columns}[T]
\begin{column}{0.43\textwidth}
\vspace{0.3em}

{\scriptsize
\begin{tabular}{@{}l c c c c@{}}
\toprule
\textbf{Cấu hình} & \textbf{A} & \textbf{B} & \textbf{C} & \textbf{D} \\
\midrule
DNABERT-2        & 81,52 & 85,56 & 87,25 & 88,36 \\
\;+ Pooling      & \textbf{83,54} & \textbf{87,63} & 89,88 & 91,08 \\
\;+ CNN          & 82,18 & 87,36 & \textbf{90,16} & 91,07 \\
PhaBERT          & 82,01 & 87,33 & 89,90 & \textbf{91,38} \\
\bottomrule
\multicolumn{5}{r}{\tiny Accuracy (\%) -- 5-fold CV}
\end{tabular}
}

\vspace{0.8em}
{\small
\begin{itemize}\setlength{\itemsep}{5pt}
    \item Pooling $\succ$ CNN ở contig ngắn (A, B)
    \item CNN $\succ$ Pooling ở contig dài (C)
    \item Kết hợp $\rightarrow$ tốt nhất ở D (\textbf{91,38\%})
    \item Sp: \textbf{+8,61\%} nhóm A (73,25$\rightarrow$81,86)
\end{itemize}
}
\end{column}
